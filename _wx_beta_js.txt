<script>
/* ══════════════════════════════════════════════════════════════════
   天气瞎报 —— 十二座城市的气象生成器（官网模式版）
   逻辑与网页版 weather.html 完全一致，只是外面套了官网的版式
   ══════════════════════════════════════════════════════════════════ */
(function () {
    'use strict';
    var REFRESH_MS = 10000;

    var CITIES = [
        { name: 'NB频道总部', ico: '🏢', base: 22, swing: 2, vol: .4,
          hum: 45, aqi: 22, vis: 28, note: '恒温机房 · 全年 22°C 上下，最稳的一座城' },
        { name: 'U星', ico: '🪐', base: 12, swing: 26, vol: 3.2,
          hum: 30, aqi: 24, vis: 42, note: '外星球 · 昼夜与四季都拉得开，一天能过完一年' },
        { name: 'AWM市', ico: '🎯', base: 6, swing: 14, vol: 1.4,
          hum: 62, aqi: 20, vis: 48, note: '高原之城 · 空气干净，风大、干爽、视野极远' },
        { name: 'Lemon市', ico: '🍋', base: 20, swing: 9, vol: 1.0,
          hum: 58, aqi: 26, vis: 26, note: '柠檬主产区 · 温暖湿润，日照给得很足' },
        { name: 'Oganesson市', ico: '☢️', base: 34, swing: 8, vol: 2.6,
          hum: 22, aqi: 38, vis: 30, note: '118 号元素城 · 全站最热情的一座，常年暖烘烘' },
        { name: 'Fafat市', ico: '🌫️', base: 17, swing: 11, vol: 1.2,
          hum: 72, aqi: 34, vis: 22, note: '水汽充沛 · 终年温润，空气里总带着点潮气' },
        { name: 'GC3市', ico: '🏗️', base: 15, swing: 12, vol: 1.1,
          hum: 52, aqi: 42, vis: 26, note: '新城 · 街宽楼新，一年一个样' },
        { name: 'UVS市', ico: '☀️', base: 25, swing: 10, vol: .9,
          hum: 34, aqi: 18, vis: 50, note: '紫外线之都 · 晴得离谱，能见度常年全站第一' },
        { name: 'UWSF市', ico: '🌬️', base: 13, swing: 15, vol: 1.6,
          hum: 55, aqi: 21, vis: 38, note: '风城 · 风常年管够，空气也一直被吹得很新' },
        { name: 'PTC市', ico: '🔥', base: 27, swing: 7, vol: 1.3,
          hum: 40, aqi: 30, vis: 28, note: '正温度系数城 · 自带加热片，冬天也不怎么冷' },
        { name: '5U市', ico: '🎈', base: 18, swing: 13, vol: 1.2,
          hum: 60, aqi: 32, vis: 25, note: '轻盈之城 · 气压常年偏低，云都飘得比别人快' },
        { name: 'Ubn市', ico: '🌃', base: 16, swing: 12, vol: 1.0,
          hum: 64, aqi: 36, vis: 24, note: '夜猫子之都 · 湿度高，晚上比白天舒服' }
    ];

    var DIRS = ['北', '东北偏北', '东北', '东北偏东', '东', '东南偏东', '东南', '东南偏南',
                '南', '西南偏南', '西南', '西南偏西', '西', '西北偏西', '西北', '西北偏北'];

    function rnd(a, b) { return a + Math.random() * (b - a); }
    function clamp(v, a, b) { return v < a ? a : (v > b ? b : v); }

    function seasonFactor(d) {
        var start = new Date(d.getFullYear(), 0, 0);
        var doy = Math.floor((d - start) / 86400000);
        return -Math.cos(2 * Math.PI * (doy - 15) / 365.25);
    }
    function seasonName(sf) {
        if (sf > .55) return '盛夏';
        if (sf > .15) return '夏末初秋';
        if (sf > -.15) return '春秋之交';
        if (sf > -.55) return '深秋初冬';
        return '隆冬';
    }
    function aqiLevel(a) {
        if (a <= 50)  return { t: '优',       c: 'wx-aq-good'  };
        if (a <= 100) return { t: '良',       c: 'wx-aq-mid'   };
        if (a <= 150) return { t: '轻度污染',  c: 'wx-aq-bad'   };
        if (a <= 200) return { t: '中度污染',  c: 'wx-aq-worse' };
        return              { t: '重度污染',  c: 'wx-aq-awful' };
    }
    function tempColor(c) {
        if (c <= -10) return '#3b82f6';
        if (c <= 0)   return '#0ea5e9';
        if (c <= 10)  return '#14b8a6';
        if (c <= 20)  return '#22c55e';
        if (c <= 28)  return '#f59e0b';
        if (c <= 35)  return '#f97316';
        return '#dc2626';
    }

    var state = CITIES.map(function (c) {
        return {
            temp: c.base + c.swing * seasonFactor(new Date()),
            aqi: c.aqi, hum: c.hum, vis: c.vis,
            dir: Math.floor(Math.random() * 16),
            force: rnd(1, 4), pres: 101.3
        };
    });

    function step(d) {
        var sf = seasonFactor(d);
        CITIES.forEach(function (c, i) {
            var s = state[i];
            var target = c.base + c.swing * sf;
            s.temp += (target - s.temp) * .35 + rnd(-c.vol, c.vol);
            s.temp = clamp(s.temp, -55, 60);

            var humTarget = c.hum - (s.temp - c.base) * 1.1;
            s.hum += (humTarget - s.hum) * .3 + rnd(-3, 3);
            s.hum = clamp(s.hum, 5, 100);

            var aqiTarget = c.aqi + (s.hum - 50) * .35 + (4 - s.force) * 5;
            s.aqi += (aqiTarget - s.aqi) * .25 + rnd(-7, 7);
            s.aqi = clamp(s.aqi, 8, 320);

            var presTarget = 101.3 - (s.temp - 15) * .085;
            s.pres += (presTarget - s.pres) * .3 + rnd(-.35, .35);
            s.pres = clamp(s.pres, 87, 108);

            var visTarget = c.vis * (1 - (s.hum - 55) / 190) - s.aqi / 45;
            s.vis += (visTarget - s.vis) * .3 + rnd(-1.2, 1.2);
            s.vis = clamp(s.vis, .2, 60);

            s.dir = (s.dir + (Math.random() < .3 ? (Math.random() < .5 ? 1 : 15) : 0)) % 16;
            s.force += rnd(-.45, .45);
            s.force = clamp(s.force, .2, 12);
        });
    }

    function windName(f) {
        if (f < .3) return { t: '无风', lv: 0 };
        if (f < 1.6) return { t: '软风', lv: 1 };
        if (f < 3.4) return { t: '轻风', lv: 2 };
        if (f < 5.5) return { t: '微风', lv: 3 };
        if (f < 8.0) return { t: '和风', lv: 4 };
        if (f < 10.8) return { t: '清风', lv: 5 };
        if (f < 13.9) return { t: '强风', lv: 6 };
        return { t: '疾风', lv: 7 };
    }

    var grid = document.getElementById('wxGrid');
    var bar = document.getElementById('tickBar');
    var num = document.getElementById('tickNum');
    var clock = document.getElementById('clockTxt');

function render() {
        var html = '';
        CITIES.forEach(function (c, i) {
            var s = state[i];
            var cT = s.temp, fT = cT * 9 / 5 + 32;
            var lv = aqiLevel(Math.round(s.aqi));
            var w = windName(s.force);
            var cd = condition(s, c);

            html +=
              '<article class="wx-card" style="--tc:' + tempColor(cT) + ';--tc-soft:' + tempColor(cT) + '1f">' +
                '<header class="wx-head">' +
                  '<span class="wx-city"><i class="wx-ico">' + c.ico + '</i>' +
                    '<b class="wx-nm">' + c.name + '</b></span>' +
                  '<span class="wx-cond"><i>' + cd.i + '</i><em>' + cd.t + '</em></span>' +
                '</header>' +
                '<p class="wx-note-in">' + c.note + '</p>' +
                '<div class="wx-temp">' +
                  '<span class="wx-c">' + cT.toFixed(1) + '</span>' +
                  '<span class="wx-u">°C</span>' +
                  '<span class="wx-f">' + fT.toFixed(1) + ' °F</span>' +
                '</div>' +
                '<dl class="wx-rows">' +
                  '<div class="wx-row"><dt>💧 湿度</dt><dd>' +
                    Math.round(s.hum) + ' %</dd></div>' +
                  '<div class="wx-row"><dt>🧭 气压</dt><dd>' +
                    s.pres.toFixed(2) + ' kPa</dd></div>' +
                  '<div class="wx-row"><dt>👁 能见度</dt><dd>' +
                    s.vis.toFixed(1) + ' km</dd></div>' +
                  '<div class="wx-row"><dt>🍃 空气</dt><dd class="' + lv.c + '">' +
                    Math.round(s.aqi) + ' ' + lv.t + '</dd></div>' +
                  '<div class="wx-row wx-row-wide"><dt>🌬 风向风力</dt><dd>' +
                    DIRS[s.dir] + '风 ' + w.lv + ' 级 · ' + w.t +
                    ' ' + s.force.toFixed(1) + ' m/s</dd></div>' +
                '</dl>' +
              '</article>';
        });
        grid.innerHTML = html;
    }

    var t0 = Date.now();
    function tickUi() {
        var el = Date.now() - t0;
        bar.style.width = Math.min(100, el / REFRESH_MS * 100).toFixed(1) + '%';
        num.textContent = Math.max(0, (REFRESH_MS - el) / 1000).toFixed(1) + 's';
    }
    function full() {
        step(new Date());
        render();
        t0 = Date.now();
        bar.style.width = '0%';
    }
    function clockUi() {
        clock.textContent = new Date().toLocaleTimeString('zh-CN', { hour12: false }) + ' 观测';
    }

    var sEl = document.getElementById('wxSeason');
    if (sEl) sEl.textContent = seasonName(seasonFactor(new Date()));
    document.getElementById('refreshNow').onclick = full;

    full(); clockUi();
    setInterval(tickUi, 100);
    setInterval(clockUi, 1000);
    setInterval(full, REFRESH_MS);
    document.addEventListener('visibilitychange', function () { if (!document.hidden) full(); });
})();
</script>
